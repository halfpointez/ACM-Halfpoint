边联通分量 e-BCC
即对当前点分量及其边构成的图来说 去掉任意一个边 任意两点依然保持联通的极大图
对于该问题 我们使用的解决方法依旧是tarjan 
tarjan的本质其实就是判断 一个点除了原先的来边 有没有别的办法能回到更早的结点

有向图中 对于一个点 如果他能回到比自己dfs序 更早的结点 那么理所当然的 这个结点和早结点间都有两条路径可以到达 所以这必然是一个边联通分量 那么对于low[u] == dfn[u] 的结点 这个结点就必然是联通的终点了

所以当我们遇到low[u] == dfn[u]时候 只需要在tarjan的基础上进行弹栈 就可以了 这就是一个联通集

#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
using i64 = long long;

i64 n, m, timer;
vector<vector<pair<i64, i64>>> Node;
vector<i64> low, dfn;
stack<i64> S;
vector<vector<i64>> ans;


void tarjan(i64 u, i64 p_id) {
	dfn[u] = low[u] = ++timer;
	S.push(u);
	for(auto [v, id]:Node[u]) {
		if(id == p_id) continue;
		if(dfn[v] == 0) {
			tarjan(v, id);
			low[u] = min(low[u], low[v]);
		} else if(dfn[v] < dfn[u]) {
			low[u] = min(low[u], dfn[v]);
		}
	}

	if(dfn[u] == low[u]) {
		vector<i64> tmp;
		while (1) {
			i64 cur = S.top();
			S.pop();
			tmp.push_back(cur);
			if(cur == u) break;
		}
		ans.push_back(tmp);
	}


	

}


int main() {
	cin >> n >> m;
	Node.resize(n + 1);
	low.assign(n + 1, 0);
	dfn.assign(n + 1, 0);
	for(int i = 1; i <= m; i++) {
		i64 u, v;
		
		cin >> u >> v;
		if(u == v) continue;
		Node[u].push_back({v, i});
		Node[v].push_back({u, i});
	}

	for(int i = 1; i <= n; i++) {
		if(!dfn[i]) tarjan(i, -1);
	}

	cout << ans.size() << endl;
	for(auto &i:ans) {
		cout << i.size() << ' ';
		for(auto &j:i) {
			cout << j << ' ';
		}
		cout << endl;
	}



	

}

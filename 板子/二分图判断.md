根据匈牙利算法 我们已经发现了 如果一个图是二分图 那么这个图上的每个相邻节点 假设有黑白两种颜色 相邻节点的颜色都步相同

所以我们可以对图进行一次bfs或者是dfs的遍历 对每个结点进行染色 然后对他的相邻结点染色相反色 当bfs完成 没有发现冲突 这个图就是一个二分图 如果发现存在冲突的 则退出 不是染色图
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    vector<int> color(n + 1, -1);
    queue<int> q;

    for (int i = 1; i <= n; i++) {
        if (color[i] != -1) continue;
        color[i] = 0;
        q.push(i);

        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int v : g[u]) {
                if (color[v] == -1) {
                    color[v] = color[u] ^ 1;
                    q.push(v);
                } else if (color[v] == color[u]) {
                    cout << -1 << '\n';
                    return 0;
                }
            }
        }
    }

    vector<int> A, B;
    for (int i = 1; i <= n; i++) {
        if (color[i] == 1) A.push_back(i);
        else B.push_back(i);  // color[i] == 0
    }

    // 输出 A
    cout << A.size() << '\n';
    for (int x : A) cout << x << ' ';
    cout << '\n';

    // 输出 B
    cout << B.size() << '\n';
    for (int x : B) cout << x << ' ';
    cout << '\n';

    return 0;
}

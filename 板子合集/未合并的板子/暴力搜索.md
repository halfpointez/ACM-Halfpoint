下面是一段从1到n全排列的输出递归代码
#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
#define endl '\n'
i64 n; // 
vector<i64> path;
vector<bool> used; 
void dfs() {
	if(path.size() == n) {
		for(auto &i:path) cout << setw(5) << i;
		cout << endl;
		return;
	}
	for(int i = 1; i <= n; i++) {
		if(used[i]) continue;
		used[i] = 1;
		path.push_back(i);
		dfs();
		used[i] = 0;
		path.pop_back();
	}
}



int main() {
	cin >> n;
	used.resize(n + 1, 0);
	dfs();
}

下面是对于组合数输出所有方案的递归代码 总共从1到n个数 选取r个
#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
#define endl '\n'
i64 n, r; //总数/选几个
vector<i64> path;
void dfs(i64 s) {
	if(path.size() == r) {
		for(auto &i:path) {
			cout << setw(3) << i;
		}
		cout << endl;
		return;
	}
	for(int i = s; i <= n; i++) {
		path.push_back(i);
		dfs(i + 1);
		path.pop_back();
	}
}

int main() {
	cin >> n >> r;
	dfs(1);
}

下面是对于一个集合中所有选择情况例子

#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
#define endl '\n'
i64 M, N, X;
i64 ans = 1e18;
vector<vector<i64>> book;
vector<i64> cost;
void dfs(i64 cur, i64 sum, vector<i64> &path) {
	if(cur == N + 1) {
		for(int i = 1; i <= M; i++) {
			if(path[i] < X) return;
		}
		ans = min(ans, sum);
		return;
	}

	dfs(cur + 1, sum, path);
	sum += cost[cur];
		for(int j = 1; j <= M; j++) path[j] += book[cur][j];
	dfs(cur + 1, sum, path);
		for(int j = 1; j <= M; j++) path[j] -= book[cur][j];

}


int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cin >> N >> M >> X;
	
	book.resize(N + 1, vector<i64>(M + 1));
	cost.resize(N + 1, 0);
	for (int i = 1; i <= N; i++) {
		cin >> cost[i];
		for (int j = 1; j <= M; j++) cin >> book[i][j];
	}
	vector<i64> path(M + 1, 0);
	dfs(1, 0, path);
	if(ans == 1e18) cout << -1;
	else
	cout << ans;
}

# 树的直径

> 所属分类: 图论 › 树上问题
> 复杂度: O(n)

## 适用问题
- 求非负权树的直径(最远两点的距离)
- 两次 BFS / 两次 DFS 求直径

// 树的直径有很多种类 对于不同的种类也有不同的求法

// 1.直接的 求树的直径 并且树的直径是非负权边(>=0)  
// 两次BFS/DFS 时间复杂度为o(n)

// 先对树上的任意一点u进行BFS/DFS 找到离这个点最远的v 在树的概念上 就是以一个点为根节点 找他的最深的叶节点
// 然后对找到的u/v再进行一次BFS/DFS 找到离v最远的点 这个距离就是树的直径 这非常的直觉 不需要怎么证明也能知道

#include <bits/stdc++.h>
using i64 = long long;
using namespace std;
#define endl '\n'

int n;
vector<vector<int>> Node;

pair<int, int> bfs(int x) { //返回最远点和距离
	vector<int> dis(n + 1, -1);
	deque<int> dq;
	int u = x;
	dq.push_back(x);
	dis[x] = 0;
	while (!dq.empty()) {
		int cur = dq.front();
		dq.pop_front();
		if (dis[cur] > dis[u]) u = cur;
		for (auto &i : Node[cur]) {
			if (dis[i] != -1) continue;
			dis[i] = dis[cur] + 1;
			dq.push_back(i);
		}
	}
	return {u, dis[u]};
}

int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);

	cin >> n;
	Node.assign(n + 1, {});
	for (int i = 1; i < n; i++) {
		int u, v;
		cin >> u >> v;
		Node[u].push_back(v);
		Node[v].push_back(u);
	}

	if (n == 1) { cout << 0 << endl; return 0; }
	
	auto [u, _] = bfs(1);
	auto [v, dis] = bfs(u);
	cout << dis << endl;
}



#include <bits/stdc++.h>
using i64 = long long;
using namespace std;
#define endl '\n'

int n;
vector<vector<int>> Node;
vector<int> dis;
int far;
int MAX;

void dfs(int cur) {
	if(dis[cur] > MAX) {
		MAX = dis[cur];
		far = cur;
	}
	for(auto &i : Node[cur]) {
		if(dis[i] != -1) continue;
		dis[i] = dis[cur] + 1;
		dfs(i);           
	}
}

void run(int start) {
	dis.assign(n + 1, -1);
	dis[start] = 0;
	MAX = -1;
	dfs(start);
}

int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);

	cin >> n;
	Node.assign(n + 1, {});
	for (int i = 1; i < n; i++) {
		int a, b;
		cin >> a >> b;
		Node[a].push_back(b);
		Node[b].push_back(a);
	}

	run(1);          
	int p = far;
	run(p);         
	cout << MAX << endl;
}

# 树上背包

> 所属分类: 动态规划 › 树形DP
> 复杂度: O(n·m)(按 siz 合并)

## 适用问题
- 树形依赖背包: 选儿子前必须先选父亲; P2014 选课(加虚拟根 0, 输出 `dp[0][M + 1]`)

## 模板

```cpp
#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
#define endl '\n'
i64 N, M;
vector<i64> course;
vector<vector<i64>> Node;
vector<vector<i64>> dp;
vector<i64> sz;

void dfs(i64 u, i64 p) {
	sz[u] += 1;
	dp[u][1] = course[u];
	for(auto &v:Node[u]) {
		if(v == p) continue;
		dfs(v, u);
		
		i64 i_max = min(sz[u] + sz[v], M + 1);
		for(i64 i = i_max; i >= 1; i--) { //总背包容量
			i64 j_max = min(sz[v], i - 1);
			for(i64 j = 0; j <= j_max; j++) {
				dp[u][i] = max(dp[u][i], dp[u][i - j] + dp[v][j]);
			}
		}

		sz[u] += sz[v];
	}
}



int main() {
	ios::sync_with_stdio(0);
	cin.tie(0);
	cin >> N >> M;	
	course.resize(N + 2);
	Node.resize(N + 2);
	dp.assign(N + 2, vector<i64>(M + 2)); 
	sz.assign(N + 2, 0);

	for(int i = 1; i <= N; i++) {
		i64 k, s;
		cin >> k >> s;
		course[i] = s;
		Node[i].push_back(k);
		Node[k].push_back(i);
	}	

	dfs(0, -1);
	cout << dp[0][M + 1];
}
```

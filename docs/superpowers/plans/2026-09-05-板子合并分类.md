# ST表(倍增)

> 所属分类: 数据结构 › 静态区间RMQ
> 复杂度: 预处理 O(n log n) / 查询 O(1)

## 适用问题
- 静态可重复贡献 RMQ: MAX、MIN、GCD、LCM、按位或、按位与

## 核心思想
ST表运用了倍增的思想: 先维护每个位置从 2^0 到 2^n(小于最后位置)长度中的RMQ。
例子: 数组 1 2 3 4 5 6 7 8, 查询 2-6 只需 2-4 和 4-6 合并。

```cpp
#include <bits/stdc++.h>
using i64 = long long;
using namespace std;
i64 n;
vector<i64> logN;
vector<i64> arr;
vector<vector<i64>> st; //修复: 原为 st(100, vector<i64>(100)) 硬编码

void pre() {
	logN[2] = 1;
	for(i64 i = 3; i <= n; i++) { //修复: 原为 i < n, logN[n] 未初始化
		logN[i] = logN[i / 2] + 1;
	}
}

void init() {
	st.assign(n, vector<i64>(logN[n] + 1)); //修复: 动态分配
	for(i64 i = 0; i < n; i++) st[i][0] = arr[i];
	for(i64 i = 1; i <= logN[n]; i++) {
		for(i64 j = 0; j + (1 << i) <= n; j++) {
			st[j][i] = max(st[j][i - 1], st[j + (1 << (i - 1))][i - 1]);
		}
	}
}

i64 query(i64 l, i64 r) { //0-based, l <= r
	i64 k = logN[r - l + 1]; //修复: 原为 l - r + 1, 下标为负越界
	return max(st[l][k], st[r - (1 << k) + 1][k]);
}

int main() { //修复: 原 main 只读入 n 未做任何事, 补全
	cin >> n;
	arr.resize(n + 1);
	logN.resize(n + 1);
	for(i64 i = 0; i < n; i++) cin >> arr[i];
	pre();
	init();
	i64 l, r;
	cin >> l >> r;
	cout << query(l, r) << endl;
}
```

## 注意
- 静态数据结构, 不支持修改;需要修改用线段树

#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
const i64 INF = 1e18;
#define endl '\n'

// 多源BFS: 无权图, 求每个点到最近源点的距离
// sources: 所有源点; dis[i] == INF 表示不可达
void multi_source_bfs(i64 n, const vector<vector<i64>> &Node, const vector<i64> &sources, vector<i64> &dis) {
	queue<i64> q;
	dis.assign(n + 1, INF);
	for(auto &s : sources) {
		dis[s] = 0;
		q.push(s);
	}
	while (!q.empty()) {
		i64 cur = q.front();
		q.pop();
		for(auto &i : Node[cur]) {
			if(dis[i] == INF) {
				dis[i] = dis[cur] + 1;
				q.push(i);
			}
		}
	}
}
```

#include <queue>

const i64 INF = 1e18;
#define endl '\n';
using namespace std;

void h_BFS(i64 n, vector<vector<i64>> &Node, vector<vector<i64>> &Good, deque<i64> &q, i64 good_now) {
	vector<bool> vis(n + 1, 0);
	vector<i64> dis(n + 1, INF);
	for(auto &i:q) {
		vis[i] = 1; //修复: 原为 0, 源点应标记已访问
		dis[i] = 0;

		i64 cur = q.front();

			if(vis[i]) continue;

			dis[i] = dis[cur] + 1;
			q.push_back(i); //修复: 原代码从不入队, BFS 不扩散
		}
	}

	for(int i = 1; i <= n; i++) {
		Good[i].push_back(dis[i]);
	}

}

# 线性筛因数

> 所属分类: 数论与数学 › 筛素数
> 复杂度: 筛 O(n) / 分解质因数 O(log n)

## 适用问题
- 筛出 1~n 每个数的最小质因子 spf
- O(log n) 分解质因数
- 由质因数分解求因数个数/约数和

## 完整模板

```cpp
#include <bits/stdc++.h>
using namespace std;
using i64 = long long;
const i64 MAXN = 1e7 + 5; //需要的范围 自行修改
vector<i64> primes; //求得的素数数组
bool is_not_prime[MAXN]; //不是素数的判断
i64 spf[MAXN]; //每个数的最小质因子

void euler_sieve(int n) {
    is_not_prime[0] = is_not_prime[1] = true; //标记0和1
    spf[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (!is_not_prime[i]) {
            primes.push_back(i);
            spf[i] = i; //质数的最小质因子是自己
        }
        for (int p : primes) {
            if ((long long)i * p > n) break;
            is_not_prime[i * p] = true;
            spf[i * p] = p;
            if (i % p == 0) break; //合数 = 最小质数 * 一个数 保证当前是最小质数 才能保证合数的唯一性
        }
    }
}

// 分解质因数 O(log n)
vector<pair<i64, i64>> factorize(i64 x) {
    vector<pair<i64, i64>> res;
    while (x > 1) {
        i64 p = spf[x];
        i64 cnt = 0;
        while (x % p == 0) {
            x /= p;
            cnt++;
        }
        res.push_back({p, cnt});
    }
    return res;
}
```
